# Profile-only raw process diagnostics; source assertions/check predicates unchanged.
import base64,datetime,types
from pathlib import Path
_DIAGNOSTIC_PATH=Path(__DIAGNOSTIC_PATH__)
_PROCESS_DIAGNOSTICS=[]
def _captured_run(args,**kwargs):
    began=datetime.datetime.now(datetime.timezone.utc).isoformat()
    try:
        observed=subprocess.run(args,capture_output=True)
    except Exception as error:
        _PROCESS_DIAGNOSTICS.append({'command':args,'started_at':began,'spawn_exception':repr(error),'returncode':None,'checker_pass':False})
        _DIAGNOSTIC_PATH.write_text(json.dumps({'scope':'Fresh profiler-process corroboration; not recovery of original empty raw child logs. Complete child bytes retained before old checker rejection.','processes':_PROCESS_DIAGNOSTICS},indent=2)+'\n')
        raise
    row={'command':args,'started_at':began,'finished_at':datetime.datetime.now(datetime.timezone.utc).isoformat(),'returncode':observed.returncode,'negative':('--negative'in args),'stdout_base64':base64.b64encode(observed.stdout).decode(),'stderr_base64':base64.b64encode(observed.stderr).decode(),'stdout_bytes':len(observed.stdout),'stderr_bytes':len(observed.stderr),'stdout_sha256':hashlib.sha256(observed.stdout).hexdigest(),'stderr_sha256':hashlib.sha256(observed.stderr).hexdigest()}
    _PROCESS_DIAGNOSTICS.append(row)
    _DIAGNOSTIC_PATH.write_text(json.dumps({'scope':'Fresh profiler-process corroboration; not recovery of original empty raw child logs. Complete child bytes retained before old checker rejection.','processes':_PROCESS_DIAGNOSTICS},indent=2)+'\n')
    return types.SimpleNamespace(returncode=observed.returncode,stdout=observed.stdout.decode('utf8').replace('\r\n','\n').replace('\r','\n'),stderr=observed.stderr.decode('utf8').replace('\r\n','\n').replace('\r','\n'))
