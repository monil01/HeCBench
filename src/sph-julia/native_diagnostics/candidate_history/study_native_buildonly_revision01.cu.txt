const double GUARD=-8191.25;
void ck(cudaError_t e){if(e!=cudaSuccess){fprintf(stderr,"CUDA:%s\n",cudaGetErrorString(e));exit(2);}}
template<class T>std::vector<T>readfile(std::string name){std::ifstream f(name,std::ios::binary|std::ios::ate);assert(f);size_t bytes=f.tellg();assert(bytes%sizeof(T)==0);std::vector<T>v(bytes/sizeof(T));f.seekg(0);f.read((char*)v.data(),bytes);assert(f);return v;}
bool fields(const double*a,const double*b,int N){double atol[14]={2e-6,5e-3,5e-9,5e-9,5e-9,2e-7,2e-7,2e-7,2e-7,2e-7,2e-7,5e-4,5e-4,5e-4};for(int i=0;i<N;i++){if(!(a[14*i]>0))return false;for(int j=0;j<14;j++){double x=a[14*i+j],y=b[14*i+j];if(!std::isfinite(x)||std::abs(x-y)>atol[j]+1e-9*std::abs(y))return false;}}return true;}
bool guards(const std::vector<double>&x,size_t n){for(size_t i=n;i<x.size();i++)if(x[i]!=GUARD)return false;return true;}
std::vector<double>capture(double*p,size_t n){std::vector<double>x(n);ck(cudaMemcpy(x.data(),p,n*8,cudaMemcpyDeviceToHost));return x;}

int main(int argc,char**argv){assert(argc==2);std::string name=argv[1];auto f=readfile<double>(name+"_initial_f64.bin"),b=readfile<double>(name+"_boundary_f64.bin"),rp=readfile<double>(name+"_params_raw.bin");assert(rp.size()==12);param p;memcpy(&p,rp.data(),96);size_t NF=f.size(),NB=b.size(),NP=rp.size();assert(NF==14*p.number_fluid_particles&&NB==6*p.number_boundary_particles);auto cpu=readfile<double>(name+"_CPU_final_f64.bin"),tracecpu=readfile<double>(name+"_CPU_checkpoints_f64.bin");std::vector<int>steps=name=="default"?std::vector<int>{1,2,10,50,100,250,500}:std::vector<int>{1,2,17};assert(tracecpu.size()==NF*steps.size());f.insert(f.end(),17,GUARD);b.insert(b.end(),17,GUARD);rp.insert(rp.end(),17,GUARD);double *df,*db,*dp;ck(cudaMalloc((void**)&df,f.size()*8));ck(cudaMalloc((void**)&db,b.size()*8));ck(cudaMalloc((void**)&dp,rp.size()*8));
auto reset=[&](){ck(cudaMemcpy(df,f.data(),f.size()*8,cudaMemcpyHostToDevice));ck(cudaMemcpy(db,b.data(),b.size()*8,cudaMemcpyHostToDevice));ck(cudaMemcpy(dp,rp.data(),rp.size()*8,cudaMemcpyHostToDevice));ck(cudaDeviceSynchronize());};

ck(cudaFree(df));ck(cudaFree(db));ck(cudaFree(dp));}
